# Global memo dictionary caching visited subtrees
def prune_tree(node, memo={}):
if not node:
return 0
if node.val in memo:
return memo[node.val] # Cache hit: O(1)
# Recursive dynamic exploration
res = node.val + prune_tree(node.left, memo)
memo[node.val] = res
return res
